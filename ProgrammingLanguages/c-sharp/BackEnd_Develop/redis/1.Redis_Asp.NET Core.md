### **1. Tích hợp Redis vào ASP.NET Core**

#### **a. Chạy Redis bằng Docker**
```
docker run --name my-redis -d -p 6379:6379 redis
```

#### **b. Cài đặt thư viện**
- Thêm package **StackExchange.Redis** vào project:
```
dotnet add package StackExchange.Redis
```

#### **c. Cấu hình Redis trong `appsettings.json`**

- Thêm cấu hình Redis:
```
{
  "Redis": {
    "ConnectionString": "localhost:6379"
  }
}
```

#### **d. Đăng ký Redis trong `Program.cs` hoặc `Startup.cs`**
- Đăng ký dịch vụ Redis:
```
using StackExchange.Redis;

var builder = WebApplication.CreateBuilder(args);

// Đăng ký Redis vào DI container
builder.Services.AddSingleton<IConnectionMultiplexer>(sp =>
    ConnectionMultiplexer.Connect(builder.Configuration["Redis:ConnectionString"]));

var app = builder.Build();
app.Run();

```

## **2. Thao tác với Redis trong ASP.NET Core**

### **📌 2.1. Lưu và lấy dữ liệu từ Redis**

Tạo một service để thao tác với Redis:
```
using StackExchange.Redis;

public class RedisService
{
    private readonly IDatabase _db;

    public RedisService(IConnectionMultiplexer redis)
    {
        _db = redis.GetDatabase();
    }

    public async Task SetAsync(string key, string value, TimeSpan? expiry = null)
    {
        await _db.StringSetAsync(key, value, expiry);
    }

    public async Task<string?> GetAsync(string key)
    {
        return await _db.StringGetAsync(key);
    }

    public async Task<bool> RemoveAsync(string key)
    {
        return await _db.KeyDeleteAsync(key);
    }
}

```

Đăng ký service vào `Program.cs`:
```
builder.Services.AddSingleton<RedisService>();
```

Sử dụng RedisService trong Controller:
```
[ApiController]
[Route("[controller]")]
public class CacheController : ControllerBase
{
    private readonly RedisService _redisService;

    public CacheController(RedisService redisService)
    {
        _redisService = redisService;
    }

    [HttpPost("set")]
    public async Task<IActionResult> SetCache(string key, string value)
    {
        await _redisService.SetAsync(key, value, TimeSpan.FromMinutes(10));
        return Ok("Đã lưu vào Redis");
    }

    [HttpGet("get")]
    public async Task<IActionResult> GetCache(string key)
    {
        var value = await _redisService.GetAsync(key);
        return Ok(value ?? "Không tìm thấy dữ liệu");
    }

    [HttpDelete("remove")]
    public async Task<IActionResult> RemoveCache(string key)
    {
        await _redisService.RemoveAsync(key);
        return Ok("Đã xóa khỏi Redis");
    }
}
```

## 🎯**3. Sử dụng Redis làm Session Storage**

Redis có thể lưu session thay vì lưu trên server bằng `DistributedCache`.

### **📌 3.1. Cài đặt Redis DistributedCache**

Chạy lệnh sau để cài đặt package:
```
dotnet add package Microsoft.Extensions.Caching.StackExchangeRedis
```

Cấu hình trong `Program.cs`:
```
builder.Services.AddStackExchangeRedisCache(options =>
{
    options.Configuration = builder.Configuration["Redis:ConnectionString"];
});

builder.Services.AddSession(options =>
{
    options.IdleTimeout = TimeSpan.FromMinutes(30);
    options.Cookie.HttpOnly = true;
    options.Cookie.IsEssential = true;
});

var app = builder.Build();
app.UseSession();
app.Run();

```

## **📌 5.2. Sử dụng Redis Session trong Controller**
```
[ApiController]
[Route("[controller]")]
public class SessionController : ControllerBase
{
    private readonly IHttpContextAccessor _httpContextAccessor;

    public SessionController(IHttpContextAccessor httpContextAccessor)
    {
        _httpContextAccessor = httpContextAccessor;
    }

    [HttpPost("set-session")]
    public IActionResult SetSession(string key, string value)
    {
        _httpContextAccessor.HttpContext.Session.SetString(key, value);
        return Ok("Session đã được lưu vào Redis");
    }

    [HttpGet("get-session")]
    public IActionResult GetSession(string key)
    {
        var value = _httpContextAccessor.HttpContext.Session.GetString(key);
        return Ok(value ?? "Session không tồn tại");
    }
}
```

## **🎯 6. Sử dụng Redis làm Cache cho Entity Framework Core**

### **📌 6.1. Cấu hình Redis Cache với EF Core**

Giả sử bạn có **ProductService** để lấy dữ liệu từ database:
```
public class ProductService
{
    private readonly ApplicationDbContext _context;
    private readonly RedisService _redisService;

    public ProductService(ApplicationDbContext context, RedisService redisService)
    {
        _context = context;
        _redisService = redisService;
    }

    public async Task<List<Product>> GetProductsAsync()
    {
        string cacheKey = "products";

        // Kiểm tra dữ liệu trong Redis trước
        var cachedProducts = await _redisService.GetAsync(cacheKey);
        if (cachedProducts != null)
        {
            return JsonSerializer.Deserialize<List<Product>>(cachedProducts);
        }

        // Nếu không có, truy vấn từ database và lưu vào Redis
        var products = await _context.Products.ToListAsync();
        await _redisService.SetAsync(cacheKey, JsonSerializer.Serialize(products), TimeSpan.FromMinutes(10));

        return products;
    }
}
```

## **🎯 7. Pub/Sub với Redis (Message Broker)**

Redis hỗ trợ **Pub/Sub** giúp các service giao tiếp với nhau theo mô hình event-driven.
```
public class RedisPubSubService
{
    private readonly IConnectionMultiplexer _redis;

    public RedisPubSubService(IConnectionMultiplexer redis)
    {
        _redis = redis;
    }

    public async Task PublishMessage(string channel, string message)
    {
        var sub = _redis.GetSubscriber();
        await sub.PublishAsync(channel, message);
    }

    public void Subscribe(string channel, Action<string> onMessageReceived)
    {
        var sub = _redis.GetSubscriber();
        sub.Subscribe(channel, (ch, msg) => onMessageReceived(msg));
    }
}
```

Sử dụng trong Controller:
```
[HttpPost("publish")]
public async Task<IActionResult> Publish(string message)
{
    await _pubSubService.PublishMessage("my-channel", message);
    return Ok("Message Sent");
}
```
