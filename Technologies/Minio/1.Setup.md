
thiết lập đơn giản:
```yml
# docker-compose.yml
services:
...
  minio:
    image: minio/minio:latest
    volumes:
      - ./minio-data:/data
    ports:
      - "9000:9000"  # API Port (Loki sẽ đẩy log vào đây)
      - "9001:9001"  # Console Port (Giao diện web để em quản lý bucket)
    environment:
      MINIO_ROOT_USER: "myaccount"
      MINIO_ROOT_PASSWORD: "mypassword"
    command: server /minio-data/data
  minio-bucket:
    image: minio/mc
    depends_on:
      - minio
    entrypoint: >
      /bin/sh -c "
      /usr/bin/mc config host add myminio http://minio:9000 minio 123@123;
      /usr/bin/mc mb myminio/file-storage;
      /usr/bin/mc policy download myminio/file-storage;
      exit 0;
      "

```
