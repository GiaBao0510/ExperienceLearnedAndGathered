Mẫu hướng dẫn cách set-up Kong API Gateway cơ bản cho Microservice. Thì Service sẽ chạy bao gồm:
- kong-database: Container PostgreSQL, là "máy chủ DB"
- Konga database prepare: Chạy 1 lần để tạo bảng cho Konga rồi thoát
- kong: Gateway chính, lưu services, routes, plugins, consumers…
- konga: Giao diện quản trị web cho Kong

```yml
version: '3.7'

services:

    #######################################
    # Postgres: The database used by Kong
    #######################################
    kong-database:
        image: postgres:18.3
        container_name: kong-database
        restart: always
        enviroment:
            POSTGRES_USER: admin
            POSTGRES_PASSWORD: gateway123
            POSTGRES_DB: kong
        ports:
            - "5433:5432"
        networks:
            - kong-net
        healthcheck:
            test: ["CMD", "pg_isready", "-U", "kong"]
            interval: 5s
            timeout: 5s
            retries: 5


    #######################################
    # Kong database migration
    #######################################
    kong-migrations:
        image: kong:3.8
        command: "kong migration bootstrap"
        restart: on-failure
        enviroment:
            KONG_DATABASE: postgres
            KONG_PG_HOST: kong-database
            KONG_PG_PASSWORD: gateway123
            KONG_PG_USER: admin
            KONG_PG_DATABASE: kong
        links:
            - kong-database
        networks:
            - kong-net
        depends_on:
            kong-database:
                condition: service_healthy
    
    #######################################
    # Kong: The API Gateway
    #######################################
    kong:
        image: kong:3.8
        container_name: kong
        restart: always
        networks:
            - kong-net
        enviroment:
            KONG_PG_HOST: kong-database
            KONG_DATABASE: postgres
            KONG_PG_USER: admin
            KONG_PG_PASSWORD: gateway123
            KONG_PG_DATABASE: kong
            KONG_PROXY_ACCESS_LOG: /dev/stdout
            KONG_ADMIN_ACCESS_LOG: /dev/stdout
            KONG_PROXY_ERROR_LOG: /dev/stderr
            KONG_ADMIN_ERROR_LOG: /dev/stderr
            KONG_PROXY_LISTEN: 0.0.0.0:8000, 0.0.0.0:8001 ssl
        depends_on:
            - kong-migrations
            - kong-database
        healthcheck:
            test: ["CMD", "curl", "-f", "http://kong:8001"]
            interval: 5
            timeout: 2s
            retries: 15
        ports:
            - "8000:8000"   # kong proxy
            - "8444:8444"   # kong admin API SSL
            - "8001:8001"   # kong admin API
            - "8443:8443"   # kong proxy SSL
    
    #######################################
    # Konga database prepare
    #######################################
    konga-prepare:
        image: pantsel/konga:next
        command: "-C prepare -a postgres -u postgresql://kong@kong-database:5432/konga_db"
        networks:
            - kong-net
        restart: on-failure
        links:
            - kong-database
        depends_on:
            - kong-database

    #######################################
    # Konga: Kong GUI
    #######################################
    konga:
        image: pantsel/konga:next
        container_name: konga
        restart: always
        networks:
            - kong-net
        enviroment:
            DB_ADAPTER: postgresql
            DB_HOST: kong-database
            DB_USER: kong
            DB_DATABASE: kong
            TOKEN_SECRET: ABCxxxxxxxxxxxxx1
        depends_on:
            - kong
        ports:
            - "1337:1337"
```
